# github.* Runtime Commands — copy-paste cheatsheet
# Fixed, compiled into the binary (internal/commands) — not YAML data.

# Convenience commands (fixed REST operation, GET, no context needed)
runtime github user get
runtime github organizations list
runtime github notifications list

# Convenience commands using the active Runtime Context's github.organization
runtime github repositories list
runtime github repositories list-for-org
runtime github issues list-for-org
runtime github teams list

# Convenience command with a JSON request body (POST) — every `key=value`
# becomes a JSON body field; true/false/numbers are coerced automatically
runtime github repositories create name=my-new-repo private=true description="created via runtime"

# Generic REST pass-through: <METHOD> <path> [key=value ...]
# GET/DELETE: key=value pairs become query params.
# POST/PUT/PATCH: key=value pairs become a JSON request body.
runtime github request GET /repos/{org}/{org}/issues state=open
runtime github request GET /user/repos per_page=5
runtime github request POST /repos/{org}/{org}/issues title="Bug" body="found via runtime"
runtime github request PATCH /repos/{org}/{org} description="updated via runtime"

# GitHub CLI (gh) — not a Runtime Command, runs via `command run` instead
# (see command_commands.txt)
runtime command run gh repo list --limit 10
runtime command run gh pr list
runtime command run gh issue create --title "Bug" --body "..."
